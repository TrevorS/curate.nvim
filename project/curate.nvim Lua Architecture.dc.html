curate.nvim / lua architecture
How the board's views map onto a Lua module tree, and the handful of real primitives that carry the weight: vim.system argv, extmark namespaces, a decoration provider, LuaCATS-typed models, and an nvr-style remote-wait for the diff-editor.

§A · THE MODULE TREE

Three layers, plus the plugin shell

Data never knows about windows; views never shell out directly. Everything mutating funnels through actions/ so undo, refresh, and the process log are uniform.

§C · jj/runner.lua — THE ONLY WAY OUT

Argv arrays through vim.system — never a shell

No string concatenation, no quoting bugs, no injection surface. Callbacks land back on the main loop via vim.schedule_wrap.

---@class curate.RunOpts
---@field cwd?  string
---@field raw?  boolean              keep bytes — templates use \x1f / \x1e
---@field env?  table<string,string>

local M = {}
local BASE = { "jj", "--no-pager", "--color=never" }

--- Async jj. The whole plugin reaches the binary through here.
---@param args     string[]                         e.g. { "log", "-r", "@" }
---@param opts     curate.RunOpts
---@param on_done  fun(r: { code:integer, stdout:string, stderr:string })
function M.spawn(args, opts, on_done)
  opts = opts or {}
  vim.system(
    vim.list_extend(vim.deepcopy(BASE), args),
    { text = not opts.raw, cwd = opts.cwd, env = opts.env },
    vim.schedule_wrap(function(r)         -- back on the main loop
      if r.code ~= 0 then require("curate.views.process").log(args, r) end
      on_done(r)
    end)
  )
end

-- one-shot blocking variant for setup/health only — never on a keypress
function M.sync(args)
  return vim.system(vim.list_extend(vim.deepcopy(BASE), args), { text = true }):wait()
end

return M
§D · jj/template.lua + model.lua — DATA, NOT TEXT

One template, control-byte separated, into typed tables

LuaCATS ---@class gives the whole plugin editor completion on the model. The graph comes from parents, computed by us — never scraped from glyphs.

local US, RS = "\31", "\30"      -- unit / record sep: bytes the data can't contain

-- field order IS the contract between template and parser
M.LOG_T = table.concat({
  "change_id", "commit_id.short(8)", "author.email()",
  "committer.timestamp().ago()", "description.first_line()",
  'parents.map(|c| c.change_id()).join(" ")',
  'if(conflict,"C","") ++ if(divergent,"D","") ++ if(immutable,"I","")',
}, ' ++ "\\u{1f}" ++ ') .. ' ++ "\\u{1e}"'

---@class curate.Change
---@field id       string      change_id — the anchor for every mark/buffer/op
---@field commit   string
---@field email    string
---@field ago      string
---@field subject  string
---@field parents  string[]    DAG edges (we lay out the graph)
---@field flags    { conflict:boolean, divergent:boolean, immutable:boolean }

---@param stdout string
---@return curate.Change[]
function M.parse_log(stdout)
  local out = {}
  for rec in vim.gsplit(stdout, RS, { plain = true }) do
    if rec ~= "" then
      local f = vim.split(rec, US, { plain = true })
      out[#out + 1] = {
        id = f[1], commit = f[2], email = f[3], ago = f[4],
        subject = f[5], parents = vim.split(f[6], " ", { trimempty = true }),
        flags = { conflict = f[7]:find("C") ~= nil,
                  divergent = f[7]:find("D") ~= nil,
                  immutable = f[7]:find("I") ~= nil },
      }
    end
  end
  return out
end
§E · ui/view.lua — THE SURFACE LIFECYCLE

One base: scratch buffer, namespace, filetype, render hook

Every view is a nofile scratch buffer with a per-kind filetype (drives ftplugin keymaps + treesitter) and its own extmark namespace. Subclasses implement only :render().

---@class curate.View
---@field kind string  @field buf integer  @field win integer  @field ns integer
local View = {}
View.__index = View

function View.new(kind)
  return setmetatable({ kind = kind,
    ns = vim.api.nvim_create_namespace("curate." .. kind) }, View)
end

function View:open(win_opts)
  self.buf = vim.api.nvim_create_buf(false, true)
  local b = vim.bo[self.buf]
  b.buftype, b.bufhidden, b.swapfile = "nofile", "wipe", false
  b.filetype = "curate-" .. self.kind     -- → ftplugin maps + TS highlight
  self.win = vim.api.nvim_open_win(self.buf, true, win_opts or { split = "right" })
  vim.wo[self.win].cursorline = true
  self:render()
  return self
end

-- refresh after any action — jj is fast, so re-render the whole buffer.
-- simplicity beats diff-patching; debounce coalesces rapid mutations.
function View:refresh()
  if self._t then self._t:stop() end
  self._t = vim.defer_fn(function() if self:valid() then self:render() end end, 80)
end

function View:valid() return self.buf and vim.api.nvim_buf_is_valid(self.buf) end
function View:render() error("subclass must implement :render()") end

return View
§F · ui/render.lua + decor.lua — HIGHLIGHTS THE MODERN WAY

Extmarks for content, a decoration provider for live state

Build {text, spans} rows, flush in one pass. Transient highlight — the rebase destination tracking your cursor — never rewrites the buffer; it's painted per-redraw by a decoration provider with ephemeral extmarks.

-- render.lua : content highlight
---@class curate.Row
---@field [1] string                  text
---@field [2]? curate.Span[]           highlights
---@alias curate.Span {[1]:string,[2]:integer,[3]:integer}

function M.flush(v, rows)
  local text = vim.tbl_map(function(r) return r[1] end, rows)
  vim.bo[v.buf].modifiable = true
  vim.api.nvim_buf_set_lines(v.buf, 0, -1, false, text)
  vim.api.nvim_buf_clear_namespace(v.buf, v.ns, 0, -1)
  for i, r in ipairs(rows) do
    for _, s in ipairs(r[2] or {}) do
      vim.api.nvim_buf_set_extmark(v.buf, v.ns,
        i - 1, s[2],
        { end_col = s[3], hl_group = s[1] })
    end
  end
  vim.bo[v.buf].modifiable = false
end
-- decor.lua : live rebase-dest highlight
-- repaints on cursor move, no re-render
local ns = vim.api.nvim_create_namespace("curate.decor")

vim.api.nvim_set_decoration_provider(ns, {
  on_line = function(_, _, buf, row)
    local st = M.state[buf]
    if not st or not st.dest then return end
    if st.row2id[row] == st.dest then
      vim.api.nvim_buf_set_extmark(buf, ns, row, 0, {
        line_hl_group = "CurateRebaseDest",
        ephemeral = true,            -- key: per-draw
      })
    end
  end,
})

-- the mark step just sets state + redraws:
function M.set_dest(buf, id)
  M.state[buf].dest = id
  vim.cmd.redraw()
end
§G · diffeditor/ — THE WEDGE, IN LUA

nvr-style remote-wait: jj blocks on a shim, the shim blocks on us

The proven pattern for "external tool edits in my running nvim and waits" — exactly how nvr --remote-wait backs $GIT_EDITOR. curate spawns jj with its servername in the env; jj spawns the shim; the shim opens the view in our instance and parks until we release it.

curate spawn jj split -i · env CURATE_SERVER=v:servername
└─> jj materializes $left / $right, runs the configured editor, blocks
└─> curate-shim $left $right (tiny client)
sockconnect(CURATE_SERVER) · rpcrequest 'curate:enter' · blocks
└─> running nvim opens DiffEditor view · user toggles hunks
user hits <CR> → write $right → reply 0 → shim exits 0 → jj records
-- actions/route.lua : launch
function M.split_interactive(change)
  jj.spawn({ "split", "-i", "-r", change.id,
    "--config", "ui.diff-editor=curate",
    "--config", 'merge-tools.curate.program=curate-shim',
    "--config", 'merge-tools.curate.edit-args=["$left","$right"]',
  }, { env = { CURATE_SERVER = vim.v.servername, CURATE_MODE = "split" } },
  function() require("curate.views.status").get():refresh() end)
end

-- diffeditor/rpc.lua : server side. The shim's rpcrequest parks here;
-- we stash its channel and reply LATER from confirm()/abort().
function M.enter(client_chan, left, right, mode)
  local v = DiffEditor.new(left, right, mode)
  v.client = client_chan                 -- who to unblock
  v:open()
  return vim.NIL                          -- ack now; real result comes via rpcnotify
end

function DiffEditor:confirm()
  self:write_back()                        -- revert de-selected hunks in $right
  vim.rpcnotify(self.client, "curate:done", 0)  -- → shim exits 0 → jj commits
  self:close()
end

function DiffEditor:abort()
  vim.rpcnotify(self.client, "curate:done", 1)  -- → exit 1 → jj cancels, no change
  self:close()
end
write_back() is the §3 protocol step 3 in Lua: for every de-selected hunk, splice that file region in $right back to its $left bytes; selected hunks stay. jj then snapshots $right. The non-interactive curate:// write-back (§H) is the same trick with cp as the whole-file tool — NicolasGB's path, generalized.
§H · views/file.lua — EDIT ANY REVISION'S FILE

curate://<change_id>/<path> via BufRead/WriteCmd

Buffers named by change-id survive rewrites and bookmark moves. Read pulls file content at the revision; write commits it back with the cp-as-merge-tool trick — no temp git repo.

-- plugin/curate.lua (registered once)
local grp = vim.api.nvim_create_augroup("curate.file", {})
for _, ev in ipairs({ "BufReadCmd", "BufWriteCmd" }) do
  vim.api.nvim_create_autocmd(ev, { group = grp, pattern = "curate://*",
    callback = function(a)
      local fn = ev == "BufReadCmd" and "read" or "write"
      require("curate.views.file")[fn](a.buf, a.match)
    end })
end

-- views/file.lua : write commits the buffer into <change> with cp-as-tool
function M.write(buf, url)
  local change, path = parse_url(url)        -- curate://CHANGE/PATH
  local tmp = vim.fn.tempname()
  vim.fn.writefile(vim.api.nvim_buf_get_lines(buf, 0, -1, false), tmp)
  jj.spawn({ "diffedit", "--restore-descendants", "-r", change,
    "--config", 'merge-tools.w.program=cp',
    "--config", ('merge-tools.w.edit-args=["%s","$right/%s"]'):format(tmp, path),
    "--tool", "w", "--", path,
  }, {}, function() vim.bo[buf].modified = false end)
end
§I · WIRING — plugin/ · setup · keymap registry · health

One registry table feeds maps, which-key, and the cheatsheet

The §2 keymap lives as data. ftplugin reads it for buffer-local <Plug> maps; which-key reads it for labels; the cheatsheet is generated from it — single source of truth.

-- plugin/curate.lua : guard + entry (once)
if vim.g.loaded_curate then return end
vim.g.loaded_curate = true

vim.api.nvim_create_user_command("Curate",
  function(a)
    local v = a.args ~= "" and a.args or "status"
    require("curate.views." .. v).open()
  end,
  { nargs = "?", complete = function()
      return { "status", "log", "oplog", "revset" }
  end })

-- lua/curate/init.lua : the public API
local M = {}
function M.setup(opts)
  require("curate.config").apply(opts or {})
  require("curate.keymap").install()
end
return M
-- keymap.lua : the registry (data, not code)
M.maps = {
  ["curate-status"] = {
    { "n", "name.new",      "new change" },
    { "d", "name.describe", "describe" },
    { "A", "route.absorb", "absorb", rw = true },
    { "s", "route.squash", "squash → parent", rw = true },
    { "o", "trust.oplog",  "op-log" },
    { "u", "trust.undo",   "undo" },
  },
}

function M.install()
  for ft, list in pairs(M.maps) do
    vim.api.nvim_create_autocmd("FileType", { pattern = ft,
      callback = function(a)
        for _, m in ipairs(list) do
          vim.keymap.set("n", m[1], function()
            local mod, f = m[2]:match("(%w+)%.(%w+)")
            require("curate.actions."..mod)[f]()
          end, { buffer = a.buf, desc = "curate: "..m[3] })
        end
      end })
  end
end
-- health.lua : :checkhealth curate
function M.check()
  vim.health.start("curate")
  local r = require("curate.jj.runner").sync({ "--version" })
  if r.code == 0 then vim.health.ok(vim.trim(r.stdout))
  else vim.health.error("jj not found on PATH") end
  if vim.v.servername ~= "" then vim.health.ok("RPC server: " .. vim.v.servername)
  else vim.health.warn("no servername — diff-editor needs a listen socket") end
end

THE SHAPE   Five primitives carry everything: the runner (argv), the parser (typed model), the View base (buffer lifecycle), the render path (extmarks + decoration provider), and the remote-wait shim (the diff-editor). Build those and every board frame is a thin :render() + a registry row on top.
